Single-line versions of D1-D5 for pasting into the Sentinel scheduled-rule box (comments stripped).

D1
let Lookback = 1h; ClaudeAudit_CL | where TimeGenerated > ago(Lookback) | where PhiLikely == true | summarize Occurrences   = count(), FirstSeen     = min(TimeGenerated), LastSeen      = max(TimeGenerated), IdentifierTypes = make_set(PiiTypes), Channels      = make_set(ChannelName), Actions       = make_set(PolicyAction), SampleQueries = make_list(QueryText, 3) by UserName, UserId | extend Severity = iff(Occurrences >= 3, "High", "Medium") | project-reorder Severity, UserName, Occurrences, IdentifierTypes, Actions

D2
let Lookback = 1h; ClaudeAudit_CL | where TimeGenerated > ago(Lookback) | where InjectionSuspected == true | summarize Attempts   = count(), Techniques = make_set(InjectionPatterns), Succeeded  = countif(Outcome == "Success"), Blocked    = countif(PolicyAction == "Blocked"), Channels   = make_set(ChannelName), Samples    = make_list(QueryText, 5), FirstSeen  = min(TimeGenerated), LastSeen   = max(TimeGenerated) by UserName, UserId, Window = bin(TimeGenerated, 15m) | extend DistinctTechniques = array_length(Techniques) | extend Severity = case(Attempts >= 3 or DistinctTechniques >= 2, "High", Succeeded > 0, "Medium", "Low") | project-reorder Severity, UserName, Attempts, DistinctTechniques, Succeeded, Blocked, Techniques

D3
let Lookback = 1h; let TZ = "America/New_York"; let BusinessStart = 7; let BusinessEnd = 20; let SensitiveChannels = dynamic(["enrollment-escalations"]); let Exempt = dynamic([]); ClaudeAudit_CL | where TimeGenerated > ago(Lookback) | where UserName !in (Exempt) | extend LocalTime = datetime_utc_to_local(TimeGenerated, TZ) | extend LocalHour = hourofday(LocalTime), Weekend = dayofweek(LocalTime) in (0d, 6d) | where LocalHour < BusinessStart or LocalHour >= BusinessEnd or Weekend | summarize Queries   = count(), FirstLocal = min(LocalTime), LastLocal  = max(LocalTime), Channels  = make_set(ChannelName), AnyPii    = max(toint(PiiDetected)), Samples   = make_list(QueryText, 3) by UserName, UserId | extend TouchedSensitive = array_length(set_intersect(Channels, SensitiveChannels)) > 0 | extend Severity = iff(TouchedSensitive or AnyPii == 1, "Medium", "Low") | project-reorder Severity, UserName, Queries, FirstLocal, Channels

D4
let Lookback = 30m; let QueryThreshold = 20; let TokenThreshold = 150000; ClaudeAudit_CL | where TimeGenerated > ago(Lookback) | where Outcome == "Success" | summarize Queries     = count(), TotalTokens = sum(InputTokens + OutputTokens), Channels    = make_set(ChannelName), Samples     = make_list(QueryText, 5), FirstSeen   = min(TimeGenerated), LastSeen    = max(TimeGenerated) by UserName, UserId, Window = bin(TimeGenerated, 30m) | where Queries >= QueryThreshold or TotalTokens >= TokenThreshold | extend QueriesPerMin = round(Queries / max_of(1.0, datetime_diff("minute", LastSeen, FirstSeen) * 1.0), 2) | project-reorder UserName, Queries, TotalTokens, QueriesPerMin, Channels

D5
let Lookback = 1h; let FileThreshold = 8; let DataFileTypes = dynamic(["csv", "xlsx", "xls", "tsv", "json", "sql"]); ClaudeAudit_CL | where TimeGenerated > ago(Lookback) | where AttachmentCount > 0 | mv-expand FileType = AttachmentTypes to typeof(string) | summarize Files         = sum(AttachmentCount), DataFiles     = countif(FileType in (DataFileTypes)), FileTypes     = make_set(FileType), Channels      = make_set(ChannelName), Samples       = make_list(QueryText, 3) by UserName, UserId, Window = bin(TimeGenerated, 1h) | where Files >= FileThreshold | extend Severity = iff(DataFiles * 2 >= Files, "High", "Medium") | project-reorder Severity, UserName, Files, DataFiles, FileTypes
